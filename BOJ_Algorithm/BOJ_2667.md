### BOJ 2667번 풀이
https://www.acmicpc.net/problem/2667

```
import sys
input = sys.stdin.readline

N = int(input())
# 지도 입력 (공백 없이 들어오므로 list(input().strip()) 활용)
graph = [list(map(int, input().strip())) for _ in range(N)]

# 상, 하, 좌, 우 이동을 위한 리스트
dx = [0, 0, 1, -1]
dy = [1, -1, 0, 0]

count = 0 # 현재 단지의 집 개수
result = [] # 각 단지의 크기를 담을 리스트

def dfs(x, y):
    global count
    graph[x][y] = 0 # 방문한 집은 0으로 바꿔서 재방문 방지
    count += 1
    
    # 4방향 탐색
    for i in range(4):
        nx = x + dx[i]
        ny = y + dy[i]
        
        # 지도 범위 안이고, 아직 집(1)이 있다면 더 깊이 탐색
        if 0 <= nx < N and 0 <= ny < N:
            if graph[nx][ny] == 1:
                dfs(nx, ny)

# 지도를 전체적으로 훑음
for i in range(N):
    for j in range(N):
        if graph[i][j] == 1:
            count = 0
            dfs(i, j)
            result.append(count) # 단지 탐색 끝난 후 크기 저장

# 정렬 후 출력
result.sort()
print(len(result)) # 총 단지 수
for r in result:
    print(r)
```
