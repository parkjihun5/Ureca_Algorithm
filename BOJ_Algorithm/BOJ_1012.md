### BOJ 1012번 풀이
https://www.acmicpc.net/problem/1012

```
import sys
# 재귀 깊이 제한을 늘려줘야 에러가 나지 않음
sys.setrecursionlimit(10000)
input = sys.stdin.readline

def dfs(x, y, m, n, graph):
    # 상, 하, 좌, 우 방향 벡터
    dx = [0, 0, 1, -1]
    dy = [1, -1, 0, 0]
    
    # 현재 배추 방문 처리 (지우기)
    graph[y][x] = 0
    
    for i in range(4):
        nx = x + dx[i]
        ny = y + dy[i]
        
        # 범위 안이고 배추가 있다면 더 깊이 탐색
        if 0 <= nx < m and 0 <= ny < n:
            if graph[ny][nx] == 1:
                dfs(nx, ny, m, n, graph)

t = int(input()) # 테스트 케이스 개수

for _ in range(t):
    m, n, k = map(int, input().split()) # 가로, 세로, 배추 개수
    graph = [[0] * m for _ in range(n)]
```
