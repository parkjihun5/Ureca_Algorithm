## BOJ 11399번 풀이
https://www.acmicpc.net/problem/11399

```
import sys

input = sys.stdin.readline

# 1. 사람 수 입력
num = int(input())

# 2. 각 사람이 걸리는 시간 입력받아 리스트로 저장
# 한 줄에 숫자가 여러 개 들어오므로 split() 후 int로 변환
numlst = list(map(int, input().split()))

# 3. 핵심: 오름차순 정렬 (빨리 끝나는 사람부터 배치)
numlst.sort()

total_sum = 0     # 모든 사람의 대기 시간 총합
current_wait = 0  # 현재 사람이 대기한 누적 시간

# 4. 리스트를 돌면서 각자의 대기 시간을 계산
for time in numlst:
    current_wait += time      # 이전 사람들의 시간 + 내 시간
    total_sum += current_wait # 각자가 느낀 대기 시간을 전체 합에 추가

# 5. 결과 출력
print(total_sum)
```

### java ver
```
import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws Exception {
        // 1. 빠른 입력을 위한 준비
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        
        // 2. 사람 수 N 입력
        int N = Integer.parseInt(br.readLine());
        
        // 3. 각 사람이 걸리는 시간을 담을 배열 생성
        int[] arr = new int[N];
        
        // 4. 한 줄에 들어온 숫자들을 공백 단위로 쪼개기
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < N; i++) {
            arr[i] = Integer.parseInt(st.nextToken());
        }
        
        // 5. 핵심: 오름차순 정렬 (작은 순서대로 줄을 서야 총 대기 시간이 최소가 됨)
        // 파이썬의 numlst.sort()와 같은 역할
        Arrays.sort(arr);
        
        int totalSum = 0;    // 모든 사람이 기다린 시간의 총합
        int currentWait = 0; // 현재 순서의 사람이 기다린 누적 시간
        
        // 6. 배열을 돌며 누적합 계산 (향상된 for문 사용)
        for (int time : arr) {
            currentWait += time;      // 이전 사람 대기 시간 + 내 업무 시간
            totalSum += currentWait;  // 총합에 누적
        }
        
        // 7. 결과 출력
        System.out.println(totalSum);
    }
}
```
