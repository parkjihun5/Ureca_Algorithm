### BOJ 1244번 풀이
https://www.acmicpc.net/problem/1244

```
import sys

input = sys.stdin.readline

switch_num = int(input())
switch_lst = list(map(int, input().split()))
student_num = int(input())

for _ in range(student_num):
    student, switch_number = map(int, input().split())

    if student == 1:
        for i in range((switch_number-1), len(switch_lst), switch_number):
            switch_lst[i] = 1 if switch_lst[i] == 0 else 0
    else:  # 여학생인 경우
        idx = switch_number - 1  # 1번 스위치 -> 0번 인덱스로 변환
        
        # 1. 일단 자기 자신은 무조건 바꾼다
        switch_lst[idx] = 1 if switch_lst[idx] == 0 else 0
        
        # 2. 양옆으로 퍼져나갈 거리 k
        k = 1
        while True:
            left = idx - k
            right = idx + k
            
            # 범위를 벗어나면 중단
            if left < 0 or right >= len(switch_lst):
                break
            
            # 양옆의 값이 같으면 둘 다 바꾸고 계속 진행
            if switch_lst[left] == switch_lst[right]:
                switch_lst[left] = 1 if switch_lst[left] == 0 else 0
                switch_lst[right] = 1 if switch_lst[right] == 0 else 0
                k += 1
            else:
                # 값이 다르면 즉시 중단
                break

        
# 결과를 20개씩 출력하는 법
# 수정된 출력 코드
for i in range(len(switch_lst)):
    if (i + 1) % 20 == 0:
        print(switch_lst[i]) # 20번째 숫자는 출력하고 줄바꿈
    else:
        if i == len(switch_lst) - 1: # 마지막 숫자라면 공백 없이 출력
            print(switch_lst[i])
        else:
            print(switch_lst[i], end=" ") # 평소에는 한 칸 띄우기
```
