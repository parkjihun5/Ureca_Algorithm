### BOJ 5397번 풀이
https://www.acmicpc.net/problem/5397

```
import sys
input = sys.stdin.readline

count = int(input())

for _ in range(count):
    password = input().strip() # strip()으로 줄바꿈 제거
    left_stack = []  # 커서 왼쪽 글자들
    right_stack = [] # 커서 오른쪽 글자들

    for i in password:
        if i == '<':
            if left_stack:
                # 왼쪽에서 꺼내 오른쪽으로 (커서가 왼쪽으로 한 칸 감)
                right_stack.append(left_stack.pop())
        elif i == '>':
            if right_stack:
                # 오른쪽에서 꺼내 왼쪽으로 (커서가 오른쪽으로 한 칸 감)
                left_stack.append(right_stack.pop())
        elif i == '-':
            if left_stack:
                # 커서 바로 왼쪽 글자 삭제
                left_stack.pop()
        else:
            # 일반 문자는 무조건 커서 왼쪽에 추가
            left_stack.append(i)

    # 결과는 [왼쪽 스택] + [오른쪽 스택을 거꾸로 뒤집은 것]
    # 오른쪽 스택은 꺼낼 때 역순으로 쌓였기 때문에 뒤집어줘야함
    print("".join(left_stack) + "".join(reversed(right_stack)))
```
