### BOJ 2178번 풀이
https://www.acmicpc.net/problem/2178

```
import sys
from collections import deque

input = sys.stdin.readline

# 1. 입력 받기 (2667번과 동일!)
N, M = map(int, input().split())
graph = [list(map(int, input().strip())) for _ in range(N)]

# 2. 방향 벡터 (상, 하, 좌, 우)
dx = [-1, 1, 0, 0]
dy = [0, 0, -1, 1]

def bfs(start_x, start_y):
    # 큐(Queue) 생성 및 시작점 추가
    queue = deque([(start_x, start_y)])
    
    while queue:
        x, y = queue.popleft()
        
        # 현재 위치에서 4방향 확인
        for i in range(4):
            nx = x + dx[i]
            ny = y + dy[i]
            
            # 미로 범위를 벗어나지 않는지 확인
            if 0 <= nx < N and 0 <= ny < M:
                # 길(1)이 있다면 탐색 진행
                if graph[nx][ny] == 1:
                    # [핵심] 다음 칸의 값 = 현재 칸의 값 + 1 (거리 누적)
                    graph[nx][ny] = graph[x][y] + 1
                    queue.append((nx, ny))
                    
    # 도착 지점(오른쪽 맨 아래)의 최단 거리 반환
    return graph[N-1][M-1]

# (0, 0)에서 출발
print(bfs(0, 0))
```
